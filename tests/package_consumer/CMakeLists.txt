cmake_minimum_required(VERSION 3.24)
project(tos_package_consumer LANGUAGES CXX)
enable_testing()

if(TOS_SOURCE_DIR)
    # A subdirectory consumer must not inherit tos warning/optimization policy.
    set(parent_flags CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_DEBUG
        CMAKE_EXE_LINKER_FLAGS CMAKE_INSTALL_RPATH CMAKE_REQUIRED_FLAGS
        CMAKE_REQUIRED_LINK_OPTIONS CMAKE_TRY_COMPILE_TARGET_TYPE)
    foreach(flag IN LISTS parent_flags)
        set(before_${flag} "${${flag}}")
    endforeach()
    add_subdirectory("${TOS_SOURCE_DIR}" tos)
    if(BUILD_TESTING OR TOS_BUILD_EXAMPLES OR TOS_INSTALL)
        message(FATAL_ERROR "tos subdirectory defaults must not enable tests, examples, or install")
    endif()
    foreach(flag IN LISTS parent_flags)
        if(NOT "${before_${flag}}" STREQUAL "${${flag}}")
            message(FATAL_ERROR "tos changed consumer build flags: ${flag}")
        endif()
    endforeach()
else()
    if(TOS_REPEAT_FIND)
        find_package(tos CONFIG REQUIRED)
        foreach(component app crypto http)
            if(TARGET tos::${component})
                message(FATAL_ERROR "default find_package imported an unrequested component")
            endif()
        endforeach()
        foreach(component IN LISTS TOS_TEST_COMPONENTS)
            find_package(tos CONFIG REQUIRED COMPONENTS ${component})
        endforeach()
    endif()
    find_package(tos ${TOS_TEST_VERSION} CONFIG REQUIRED
        COMPONENTS ${TOS_TEST_COMPONENTS} OPTIONAL_COMPONENTS ${TOS_TEST_OPTIONAL})
    foreach(component IN LISTS TOS_TEST_OPTIONAL)
        if(tos_${component}_FOUND OR TARGET tos::${component})
            message(FATAL_ERROR "unavailable optional component ${component} was reported found")
        endif()
    endforeach()
endif()

get_target_property(base_compile_options tos::base INTERFACE_COMPILE_OPTIONS)
if(base_compile_options MATCHES "fsanitize|fno-omit-frame-pointer|^-O")
    message(FATAL_ERROR "tos leaked instrumentation/optimization into consumer compilation")
endif()

foreach(component IN LISTS TOS_TEST_ABSENT)
    if(TARGET tos::${component})
        message(FATAL_ERROR "disabled/unrequested component ${component} was imported")
    endif()
endforeach()
get_target_property(base_links tos::base INTERFACE_LINK_LIBRARIES)
if(base_links MATCHES "OpenSSL|CURL|tos(::)?(crypto|http|app)")
    message(FATAL_ERROR "base has unwanted dependencies: ${base_links}")
endif()
if(TARGET tos::http)
    get_target_property(http_links tos::http INTERFACE_LINK_LIBRARIES)
    if(http_links MATCHES "OpenSSL|tos(::)?crypto")
        message(FATAL_ERROR "HTTP directly depends on crypto: ${http_links}")
    endif()
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE tos::base)
foreach(component IN LISTS TOS_TEST_COMPONENTS)
    target_link_libraries(consumer PRIVATE tos::${component})
    string(TOUPPER "${component}" upper)
    target_compile_definitions(consumer PRIVATE TOS_TEST_${upper})
endforeach()
add_test(NAME consumer COMMAND consumer)
set_tests_properties(consumer PROPERTIES TIMEOUT 30)
